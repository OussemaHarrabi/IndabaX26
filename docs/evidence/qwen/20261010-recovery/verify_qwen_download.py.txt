"""Verify own Kaggle evidence without recomputing or altering outcomes."""
import argparse
import hashlib
import json
import pathlib
import stat
import sys
import zipfile


def extract_checked(archive, destination):
    destination = pathlib.Path(destination).resolve()
    with zipfile.ZipFile(archive) as z:
        corrupt = z.testzip()
        if corrupt:
            raise ValueError('Corrupt archive entry: ' + corrupt)
        for info in z.infolist():
            path = pathlib.PurePosixPath(info.filename)
            if path.is_absolute() or '..' in path.parts or '\\' in info.filename or ':' in info.filename:
                raise ValueError('Unsafe archive path')
            if stat.S_ISLNK(info.external_attr >> 16):
                raise ValueError('Archive symlink is unsupported')
            target = (destination / info.filename).resolve()
            if not target.is_relative_to(destination):
                raise ValueError('Archive path escaped destination')
        z.extractall(destination)


def verify(root, repo, expected_source):
    sys.path.insert(0, str(repo / 'backend'))
    sys.path.insert(0, str(repo))
    from benchmark.campaign import verify_hashes
    rows = []
    for run in sorted((root / 'runs').iterdir()):
        if not run.is_dir():
            continue
        manifest = json.loads((run / 'manifest.json').read_text(encoding='utf-8'))
        campaign = manifest['campaign']
        checkpoints = list((run / 'checkpoints').glob('*.json'))
        hashes = verify_hashes(run)
        identity = manifest.get('model', {})
        rows.append({
            'run': run.name, 'source_commit': manifest['code']['commit'],
            'source_matches': manifest['code']['commit'] == expected_source,
            'source_dirty': manifest['code'].get('dirty'),
            'stage': campaign['stage'], 'seed': campaign['seed'], 'condition': campaign['condition'],
            'status': campaign['status'], 'recorded_checkpoints': len(checkpoints),
            'declared_total': campaign['scenarios_total'], 'completed': campaign['scenarios_completed'],
            'errored': campaign['scenarios_errored'], 'remaining': campaign['scenarios_remaining'],
            'hash_verification': hashes, 'model': identity,
        })
    errors = []
    if not rows:
        errors.append('no_run_manifests')
    for row in rows:
        if not row['source_matches'] or row['source_dirty'] or not row['hash_verification']['ok']:
            errors.append(row['run'] + ':integrity_or_source_failure')
        if row['recorded_checkpoints'] != row['declared_total'] or row['remaining'] or row['status'] != 'complete':
            errors.append(row['run'] + ':incomplete')
    # Scenario errors remain visible. They are never turned into protection.
    return {'version': 'qwen-evidence-download-verification/v1', 'integrity_and_completeness_ok': not errors,
            'errors': errors, 'runs': rows, 'total_checkpoints': sum(r['recorded_checkpoints'] for r in rows),
            'total_errored_episodes': sum(r['errored'] for r in rows),
            'general_asr': None, 'general_bts': None,
            'limitation': 'Integrity/completeness only; original positional scores are not semantically validated.'}


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--root', type=pathlib.Path, required=True)
    parser.add_argument('--repo', type=pathlib.Path, required=True)
    parser.add_argument('--source', required=True)
    parser.add_argument('--archive', type=pathlib.Path)
    args = parser.parse_args()
    if args.archive:
        extract_checked(args.archive, args.root)
        print('archive_sha256', hashlib.sha256(args.archive.read_bytes()).hexdigest())
    result = verify(args.root, args.repo, args.source)
    (args.root / 'download-verification.json').write_text(json.dumps(result, indent=2) + '\n', encoding='utf-8')
    print(json.dumps({k: v for k, v in result.items() if k != 'runs'}, indent=2))
    if not result['integrity_and_completeness_ok']:
        raise SystemExit(2)
